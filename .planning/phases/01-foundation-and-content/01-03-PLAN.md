  <files>
    src/schemas/product.schema.ts

  <action>
    Create Zod schemas with TypeScript type inference:

    1. src/schemas/product.schema.ts:
       ```typescript
       import { z } from 'zod';

       export const productSchema = z.object({
         id: z.string().min(1),
         name: z.string().min(1).max(100),
         description: z.string().max(500),
         price: z.number().positive(),
         category: z.enum(['snack', 'sweet', 'savory']),
         imageUrl: z.string().url().optional(),
         dietary: z.array(z.enum(['vegetarian', 'vegan', 'gluten-free', 'dairy-free', 'nut-free'])).default([]),
         allergens: z.array(z.string()).default([]),
         available: z.boolean().default(true)
       });

       export const productsSchema = z.array(productSchema);
       export type Product = z.infer<typeof productSchema>;
       ```

    2. src/schemas/meal.schema.ts:
       ```typescript
       import { z } from 'zod';

       export const mealSchema = z.object({
         id: z.string().min(1),
         name: z.string().min(1).max(100),
         description: z.string().max(500),
         price: z.number().positive(),
         imageUrl: z.string().url().optional(),
         servings: z.string().optional(), // e.g., "Serves 2-3"
         dietary: z.array(z.enum(['vegetarian', 'vegan', 'gluten-free', 'dairy-free', 'nut-free'])).default([]),
         allergens: z.array(z.string()).default([]),
         available: z.boolean().default(true),
         weekOf: z.string().optional() // e.g., "2026-01-27" - week start date
       });

       export const mealsSchema = z.array(mealSchema);
       export type Meal = z.infer<typeof mealSchema>;
       ```

    Key design decisions:
    - id is string (not UUID) for admin simplicity - they can use "biltong-sticks" style slugs
    - imageUrl is optional (admin may not have image yet)
    - dietary array allows multiple tags
    - available flag lets admin hide items without deleting
    - weekOf for meals helps track which week's menu
  </action>
  <verify>

    Expected: No TypeScript errors
  </verify>
  <done>
    Zod schemas compile without errors. Schemas export both runtime validators and TypeScript types.

<task type="auto">
  <name>Task 3: Create Content Loading Utilities</name>
  <files>
    src/lib/content.ts
  </files>
  <action>
    Create utility functions that load and validate JSON content:

    ```typescript
    import { productsSchema, type Product } from '../schemas/product.schema';
    import { mealsSchema, type Meal } from '../schemas/meal.schema';
    import productsData from '../content/products.json';
    import mealsData from '../content/meals.json';

    // Validate at module load time (build time for Astro static)
    // If validation fails, build fails with clear error message
    export const products: Product[] = productsSchema.parse(productsData);
    export const meals: Meal[] = mealsSchema.parse(mealsData);

    // Helper functions for common queries
    export function getProducts(): Product[] {
      return products.filter(p => p.available);
    }

    export function getMeals(): Meal[] {
      return meals.filter(m => m.available);
    }

    export function getProductById(id: string): Product | undefined {
      return products.find(p => p.id === id);
    }

    export function getProductsByCategory(category: Product['category']): Product[] {
      return products.filter(p => p.available && p.category === category);
    }
    ```

    The key insight: validation happens at import time. When Astro builds the site, it imports this module, which parses the JSON. Invalid JSON = build failure with Zod error message showing exactly what's wrong.
